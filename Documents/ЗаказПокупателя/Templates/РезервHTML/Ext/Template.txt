<script>
(function(){
  var modal=document.getElementById("imageModal");
  if(!modal)return;

  var modalImage=modal.querySelector(".image-modal__image");
  var modalClose=modal.querySelector(".image-modal__close");
  var productLink=modal.querySelector(".image-modal__product");
  var lastFocusedElement=null;

  function openModal(button){
    var imageUrl=button.getAttribute("data-full-image");
    var productUrl=button.getAttribute("data-product-url");

    if(!imageUrl)return;

    lastFocusedElement=button;
    modalImage.setAttribute("src",imageUrl);

    if(productUrl){
      productLink.setAttribute("href",productUrl);
      productLink.style.display="inline-block";
    }else{
      productLink.removeAttribute("href");
      productLink.style.display="none";
    }

    modal.classList.add("is-open");
    document.body.style.overflow="hidden";
    modalClose.focus();
  }

  function closeModal(){
    modal.classList.remove("is-open");
    modalImage.setAttribute("src","");
    document.body.style.overflow="";

    if(lastFocusedElement)lastFocusedElement.focus();
  }

  document.querySelectorAll(".thumb-button").forEach(function(button){
    button.addEventListener("click",function(){
      openModal(button);
    });
  });

  modalClose.addEventListener("click",closeModal);

  modal.addEventListener("click",function(event){
    if(event.target===modal)closeModal();
  });

  modalImage.addEventListener("error",closeModal);

  document.addEventListener("keydown",function(event){
    if(event.key==="Escape"&&modal.classList.contains("is-open"))closeModal();
  });
})();

document.querySelectorAll(".tab").forEach(function(btn){
  btn.addEventListener("click",function(){
    document.querySelectorAll(".tab").forEach(function(button){
      button.classList.remove("is-active");
    });

    document.querySelectorAll(".tabpanel").forEach(function(panel){
      panel.classList.remove("is-active");
    });

    btn.classList.add("is-active");

    var panel=document.getElementById(btn.getAttribute("data-tab"));
    if(panel)panel.classList.add("is-active");
  });
});

document.querySelectorAll(".copyable").forEach(function(el){
  el.setAttribute("tabindex","0");
  el.setAttribute("role","button");

  function copied(){
    el.classList.add("copied");
    setTimeout(function(){
      el.classList.remove("copied");
    },1400);
  }

  function fallbackCopy(value){
    var area=document.createElement("textarea");
    area.value=value;
    area.style.position="fixed";
    area.style.opacity="0";
    document.body.appendChild(area);
    area.focus();
    area.select();

    try{
      document.execCommand("copy");
      copied();
    }catch(e){}

    document.body.removeChild(area);
  }

  function copy(){
    var value=el.getAttribute("data-copy");
    if(!value)return;

    if(navigator.clipboard&&window.isSecureContext){
      navigator.clipboard.writeText(value).then(copied).catch(function(){
        fallbackCopy(value);
      });
    }else{
      fallbackCopy(value);
    }
  }

  el.addEventListener("click",copy);

  el.addEventListener("keydown",function(event){
    if(event.key==="Enter"||event.key===" "){
      event.preventDefault();
      copy();
    }
  });
});
